// lib/turn-state.ts
export type TurnState =
  | "idle" | "starting" | "working" | "waiting"
  | "stopping" | "interrupted" | "failed" | "reconnecting";

export interface TurnInputs {
  running: boolean; busy: boolean; status: string; slowReconnect: boolean;
  retrying: boolean; pending: PendingRequest[]; goal: Goal | null;
  failure: TurnFailure | null; stopRequested: boolean;
  lastTurnEnd: "completed" | "interrupted" | null;
}
export interface TurnView {
  state: TurnState;
  label: string;           // pill label from the table
  indicator: "dot" | "spinner";
  tone: "idle" | "accent" | "warn" | "bad";
  pulse: boolean;          // false when prefers-reduced-motion
  frozen: boolean;         // reconnecting: counts hold their last value
}

export function deriveTurnState(i: TurnInputs): TurnView {
  if (i.status === "Reconnecting" || i.status === "Disconnected") return reconnecting(i);
  if (i.failure && !i.retrying) return view("failed");
  if (i.running && i.stopRequested) return view("stopping");
  if (i.running && i.pending.length > 0) return view("waiting");
  if (i.running) return working(i);          // goal / retry label
  if (i.busy) return view("starting");
  if (i.lastTurnEnd === "interrupted") return view("interrupted");
  return view("idle");
}

Esc order (fixed)

Modal layers handle their own Esc first and stop propagation: palette, Sheet, AgentDialog, help sheet. Everything else goes through one window handler in lib/escape-stack.ts, which does the first step that applies and stops:

  1. Close the slash or @ menu.
  2. Exit goal mode.
  3. Close the open popover or menu, then the open panel (activity panel, or the sidebar drawer at ≤720).
  4. Stop the running turn, only when running and nothing above applied.
  • Step 4 runs stopTurn() and puts "Stopping" in the pill. It never runs on key repeat (e.repeat), so holding Esc can't stop a turn.
  • D-06: when focus is inside a docked request card, Esc after steps 1 to 3 cancels that request instead of stopping the turn.
  • Inputs that own Esc (NameInput, Share link field) handle it and stop propagation.
// lib/escape-stack.ts
type EscStep = { id: "menu" | "goal" | "overlay" | "request" | "stop"; active: () => boolean; run: () => void };
export function handleEscape(e: KeyboardEvent, steps: EscStep[]): boolean {
  if (e.key !== "Escape" || e.defaultPrevented || e.isComposing) return false;
  const step = steps.find(s => s.active());
  if (!step || (step.id === "stop" && e.repeat)) return false;
  e.preventDefault(); step.run(); return true;
}
